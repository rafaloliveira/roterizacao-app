# Função de leitura e preparação dos dados (adaptada para base64)
def load_and_prepare_data(base64_file_content, filename):
    if base64_file_content is None:
        return {'status': 'warning', 'message': 'Nenhum arquivo enviado.'}

    try:
        content_type, content_string = base64_file_content.split(',')
        decoded = base64.b64decode(content_string)

        # Assumindo que o arquivo é um XLSX
        df = pd.read_excel(io.BytesIO(decoded), sheet_name="Sheet1") # O nome da sheet pode ser uma constante

        # Garante que as colunas críticas existam e sejam strings antes de qualquer operação
        if "Serie/Numero CTRC" in df.columns:
            df.rename(columns={"Serie/Numero CTRC": "Serie_Numero_CTRC"}, inplace=True)
        if "Serie_Numero_CTRC" not in df.columns:
            return {'status': 'error', 'message': "Coluna 'Serie/Numero CTRC' não encontrada no arquivo."}
        df["Serie_Numero_CTRC"] = df["Serie_Numero_CTRC"].astype(str)

        # Forçar TODAS as colunas de data que vêm do Excel para tipo string IMEDIATAMENTE após a leitura.
        for col_data_excel in colunas_data:
            if col_data_excel in df.columns:
                df[col_data_excel] = df[col_data_excel].astype(str)
            #else: # Não é um erro crítico se a coluna não existe no Excel, apenas um aviso
            #    print(f"DEBUG: Coluna de data '{col_data_excel}' não encontrada no Excel.")

        # O restante da função load_and_prepare_data continua como estava, mas adaptando prints para retornos
        excel_to_supabase_col_map = {
            "Cubagem em m3": "Cubagem em m³",
            "Serie/Numero CTRC": "Serie_Numero_CTRC"
        }
        df.rename(columns=excel_to_supabase_col_map, inplace=True)

        supabase_columns = [
            "Serie_Numero_CTRC", "Serie/Numero CT-e", "Tipo do Documento", "Unidade Emissora",
            "Data de Emissao", "Data de Autorizacao", "Chave CT-e", "Cliente Remetente",
            "UF do Remetente", "UF do Expedidor", "Cliente Pagador", "UF do Pagador",
            "Fone do Pagador", "Segmento do Pagador", "CNPJ Destinatario", "Cliente Destinatario",
            "Bairro do Destinatario", "Setor de Destino", "UF do Destinatario", "Local de Entrega",
            "Bairro", "Cidade de Entrega", "UF de Entrega", "Unidade Receptora",
            "Numero da Nota Fiscal", "Peso Real em Kg", "Cubagem em m³", "Quantidade de Volumes",
            "Valor da Mercadoria", "Valor do Frete", "Valor do ICMS", "Valor do ISS",
            "Peso Calculado em Kg", "Frete Peso", "Frete Valor", "TDA", "TDE",
            "Adicional de Frete", "UF origem da prestacao", "Codigo da Ultima Ocorrencia",
            "Data de inclusao da Ultima Ocorrencia", "Data da Ultima Ocorrencia",
            "Usuario da Ultima Ocorrencia", "Unidade da Ultima Ocorrencia",
            "Descricao da Ultima Ocorrencia", "Latitude da Ultima Ocorrencia",
            "Longitude da Ultima Ocorrencia", "Previsao de Entrega", "Entrega Programada",
            "Data da Entrega Realizada", "Quantidade de Dias de Atraso", "Localizacao Atual",
            "Data do Cancelamento", "Motivo do Cancelamento", "Codigo dos Correios",
            "Numero da Capa de Remessa", "Numero do Pacote de Arquivamento",
            "Compr. de Entrega Escaneado", "Data do Escaneamento", "Hora do Escaneamento",
            "Notas Fiscais", "Numero dos Pedidos", "Chaves NF-es",
            "Volume Cliente/Shipment", "Unnamed: 67","CEP de Entrega","CEP do Destinatario"
        ]

        # Use uma lista para evitar alterar a lista original (colunas_texto)
        internal_colunas_texto = list(colunas_texto)
        internal_colunas_numero = list(colunas_numero)
        internal_colunas_data = list(colunas_data)

        df = corrigir_tipos(df, internal_colunas_texto, internal_colunas_numero, internal_colunas_data)

        seen = set()
        final_columns = []
        for col in supabase_columns:
            if col in df.columns and col not in seen:
                final_columns.append(col)
                seen.add(col)
        df = df[final_columns]

        # Conversão numérica e tratamento de booleanos (mantido do original)
        for col in internal_colunas_numero:
            if col in df.columns:
                df[col] = df[col].apply(lambda x: str(x).replace(',', '.').strip() if pd.notnull(x) else None)
                df[col] = pd.to_numeric(df[col], errors='coerce')
                # if col in int_cols: # int_cols não está definido como global, removido.
                #     df[col] = df[col].astype('Int64')

        boolean_column = "Compr. de Entrega Escaneado"

        df = df.replace({np.nan: None, pd.NaT: None, pd.NA: None})

        primary_key = "Serie_Numero_CTRC"
        if primary_key in df.columns and df[primary_key].isnull().any():
            df.dropna(subset=[primary_key], inplace=True)
            if df.empty:
                 return {'status': 'error', 'message': f"Nenhuma entrega válida após remover nulos na chave primária '{primary_key}'."}


        if "CNPJ Destinatario" in df.columns:
            df["CNPJ Destinatario"] = df["CNPJ Destinatario"].astype(str).str.strip()

        data_to_insert = df.to_dict(orient='records')

        # Retorna o DataFrame como lista de dicionários para ser JSON-serializável
        return {'status': 'success', 'message': f"Arquivo '{filename}' lido com sucesso.", 'data': data_to_insert}

    except Exception as e:
        return {'status': 'error', 'message': f"Erro ao processar o Excel: {e}"}

def corrigir_tipos(df, colunas_texto_list, colunas_numero_list, colunas_data_list):
    # Converter para texto (string)
    for col in colunas_texto_list:
        if col in df.columns:
            df[col] = df[col].astype(str).replace({'nan': None, 'NaT': None})

    # Converter para numérico
    for col in colunas_numero_list:
        if col in df.columns:
            df[col] = pd.to_numeric(df[col], errors='coerce')

     # Converter para datetime
    for col in colunas_data_list:
        if col in df.columns:
            df[col] = df[col].astype(str).str.replace('/', '-', regex=False)
            df[col] = _parse_date_robustly(df[col], col_name=col)
    return df

def sincronizar_fluxo_com_fbaseroter(df_fbaseroter_active_prepared_list):
    """
    Sincroniza as tabelas do fluxo de trabalho (`FLOW_TABLES`) com a tabela `fBaseroter`.
    Recebe `df_fbaseroter_active_prepared_list` como lista de dicionários.
    Retorna um dicionário com o status da sincronização e as contagens de atualizações.
    """
    df_fbaseroter_active_prepared = pd.DataFrame(df_fbaseroter_active_prepared_list)

    if df_fbaseroter_active_prepared.empty:
        return {'status': 'warning', 'message': "fBaseroter está vazia. Nenhuma entrega no fluxo será atualizada ou adicionada via sincronização de fluxo.", 'updated_counts': {}}

    if 'Serie_Numero_CTRC' not in df_fbaseroter_active_prepared.columns:
        return {'status': 'error', 'message': "Erro crítico: 'Serie_Numero_CTRC' não encontrada no DataFrame preparado da fBaseroter. A sincronização será interrompida.", 'updated_counts': {}}

    active_fbaseroter_ctrcs = set(df_fbaseroter_active_prepared['Serie_Numero_CTRC'].tolist())
    updated_in_flow_tables = {}

    for table_name in FLOW_TABLES:
        try:
            response_flow_table = supabase.table(table_name).select("Serie_Numero_CTRC").execute()
            df_flow_table_existing_ctrcs = pd.DataFrame(response_flow_table.data)

            current_flow_ctrcs = set()
            if not df_flow_table_existing_ctrcs.empty:
                current_flow_ctrcs = set(df_flow_table_existing_ctrcs['Serie_Numero_CTRC'].astype(str).tolist())
            
            # --- Lógica para Deletar registros do fluxo que não estão mais em fBaseroter ---
            ctrcs_to_delete_from_flow_table = current_flow_ctrcs - active_fbaseroter_ctrcs
            if ctrcs_to_delete_from_flow_table:
                supabase.table(table_name).delete().in_("Serie_Numero_CTRC", list(ctrcs_to_delete_from_flow_table)).execute()
                #print(f"DEBUG: {len(ctrcs_to_delete_from_flow_table)} registro(s) deletado(s) de '{table_name}'.")

            # --- Lógica para Atualizar/Inserir registros existentes e novos no fluxo ---
            ctrcs_to_update_or_insert = list(active_fbaseroter_ctrcs) # Todos os CTRCs ativos em fBaseroter
            
            if ctrcs_to_update_or_insert:
                df_updates = df_fbaseroter_active_prepared[
                    df_fbaseroter_active_prepared['Serie_Numero_CTRC'].isin(ctrcs_to_update_or_insert)
                ].copy()
                
                # Filter df_updates to only include columns that exist in the target table in Supabase
                # This is a safety measure to prevent errors from trying to upsert columns that don't exist
                target_table_columns = supabase.rpc('get_table_columns', {'p_table_name': table_name}).execute().data # Assumes you have an RPC function to get table columns
                
                if target_table_columns:
                    target_columns_set = {col['column_name'] for col in target_table_columns}
                    df_updates = df_updates[[col for col in df_updates.columns if col in target_columns_set]]
                else:
                    print(f"WARNING: Could not fetch column names for table '{table_name}'. Proceeding with all columns from fBaseroter.")


                records_to_update = df_updates.to_dict(orient='records')
                records_to_update = _prepare_df_for_supabase_insert(pd.DataFrame(records_to_update)).to_dict(orient='records') # Ensure dates are formatted correctly

                if records_to_update:
                    response = supabase.table(table_name).upsert(records_to_update, on_conflict='Serie_Numero_CTRC').execute() # Use on_conflict for upsert
                    if hasattr(response, 'data') and response.data:
                        updated_in_flow_tables[table_name] = len(response.data)
                        #print(f"DEBUG: {len(response.data)} entrega(s) atualizada(s)/inserida(s) em '{table_name}'.")

        except Exception as e:
            print(f"Erro ao sincronizar tabela '{table_name}': {e}") # Print para Dash
            # Não retorna erro crítico aqui para não parar o processo inteiro, apenas registra.

    return {'status': 'success', 'message': "Sincronização do fluxo de trabalho com fBaseroter concluída.", 'updated_counts': updated_in_flow_tables}

def aplicar_regras_e_preencher_tabelas():
    """
    Aplica regras de negócio para mover entregas da `fBaseroter` para
    `pre_roterizacao` ou `confirmadas_producao`.
    Retorna o número de entregas inseridas em cada tabela.
    """
    try:
        df_raw_data = supabase.table("fBaseroter").select("*").execute().data
        if not df_raw_data:
            return {'status': 'warning', 'message': "Tabela fBaseroter está vazia. Nenhuma entrega será processada.", 'inserted_prerot': 0, 'inserted_confirm': 0}

        df = pd.DataFrame(df_raw_data)
        df.columns = df.columns.str.strip()

        workflow_tables = [
            "confirmadas_producao", "aprovacao_diretoria", "pre_roterizacao",
            "cargas_geradas", "aprovacao_custos", "cargas_aprovadas"
        ]
        existing_workflow_ctrcs = set()
        for table_name in workflow_tables:
            try:
                response = supabase.table(table_name).select("Serie_Numero_CTRC").execute()
                current_ctrcs = {str(record["Serie_Numero_CTRC"]) for record in response.data}
                existing_workflow_ctrcs.update(current_ctrcs)
            except Exception as e:
                print(f"WARNING: Não foi possível ler CTRCs da tabela '{table_name}': {e}.")

        # Filtra o DataFrame para incluir APENAS entregas que AINDA NÃO estão no fluxo de trabalho
        df_novas_para_fluxo = df[~df['Serie_Numero_CTRC'].isin(existing_workflow_ctrcs)].copy()

        if df_novas_para_fluxo.empty:
            return {'status': 'info', 'message': "Nenhuma nova entrega encontrada para ser inserida no fluxo.", 'inserted_prerot': 0, 'inserted_confirm': 0}

        df_processar = df_novas_para_fluxo
        df_processar['Previsao de Entrega'] = pd.to_datetime(df_processar.get('Previsao de Entrega'), errors='coerce')
        df_processar['Entrega Programada'] = pd.to_datetime(df_processar.get('Entrega Programada'), errors='coerce')

        # --- Merge com Micro_Regiao_por_data_embarque ---
        micro_data = supabase.table("Micro_Regiao_por_data_embarque").select("*").execute().data
        if micro_data:
            df_micro = pd.DataFrame(micro_data)
            df_micro.columns = df_micro.columns.str.strip()
            col_data_micro = [col for col in df_micro.columns if 'relação' in col.lower()]
            cidade_destino_col = 'CIDADE DESTINO'
            regiao_col = 'REGIÃO'
            if col_data_micro and cidade_destino_col in df_micro.columns and regiao_col in df_micro.columns:
                data_col = col_data_micro[0]
                df_micro[data_col] = pd.to_numeric(df_micro[data_col], errors='coerce')
                df_processar['Cidade de Entrega_upper'] = df_processar['Cidade de Entrega'].astype(str).str.strip().str.upper()
                df_micro[cidade_destino_col + '_upper'] = df_micro[cidade_destino_col].astype(str).str.strip().str.upper()
                df_processar = df_processar.merge(
                    df_micro[[data_col, cidade_destino_col + '_upper', regiao_col]],
                    how='left',
                    left_on='Cidade de Entrega_upper',
                    right_on=cidade_destino_col + '_upper'
                )
                df_processar.rename(columns={regiao_col: 'Regiao'}, inplace=True)
                df_processar['Data de Embarque'] = df_processar['Previsao de Entrega'] - pd.to_timedelta(df_processar[data_col], unit='D')
                df_processar.drop(columns=[data_col, cidade_destino_col + '_upper', 'Cidade de Entrega_upper'], inplace=True, errors='ignore')
            else:
                df_processar['Data de Embarque'] = pd.NaT
                df_processar['Regiao'] = None
        else:
            df_processar['Data de Embarque'] = pd.NaT
            df_processar['Regiao'] = None

        # --- Merge com Particularidades ---
        part_data = supabase.table("Particularidades").select("*").execute().data
        if part_data:
            df_part = pd.DataFrame(part_data)
            df_part.columns = df_part.columns.str.strip()
            df_processar = df_processar.merge(df_part[['CNPJ', 'Particularidade']], how='left',
                          left_on='CNPJ Destinatario', right_on='CNPJ', suffixes=('', '_part'))
            df_processar['Particularidade'] = df_processar['Particularidade_part'].fillna(df_processar['Particularidade'])
            df_processar.drop(columns=['CNPJ_part', 'Particularidade_part'], inplace=True, errors='ignore')
        else:
            df_processar['Particularidade'] = None

        # --- Merge com Clientes_Entrega_Agendada ---
        agendados_data = supabase.table("Clientes_Entrega_Agendada").select("*").execute().data
        if agendados_data:
            df_ag = pd.DataFrame(agendados_data)
            df_ag.columns = df_ag.columns.str.strip()
            if 'CNPJ' in df_ag.columns and 'Status de Agenda' in df_ag.columns:
                cnpjs_agendar = df_ag[df_ag['Status de Agenda'].astype(str).str.upper() == 'AGENDAR']['CNPJ'].astype(str).str.strip().unique()
                df_processar['Status'] = df_processar['CNPJ Destinatario'].astype(str).str.strip().isin(cnpjs_agendar).map({True: 'AGENDAR', False: None})
            else:
                df_processar['Status'] = None
        else:
            df_processar['Status'] = None

        # --- Definição da Rota ---
        df_processar['Rota'] = None
        rotas_data = supabase.table("Rotas").select("*").execute().data
        df_rotas = pd.DataFrame(rotas_data) if rotas_data else pd.DataFrame()
        df_rotas.columns = df_rotas.columns.str.strip()
        rotas_poas_data = supabase.table("RotasPortoAlegre").select("*").execute().data
        df_poas = pd.DataFrame(rotas_poas_data) if rotas_poas_data else pd.DataFrame()
        df_poas.columns = df_poas.columns.str.strip()

        for idx, row in df_processar.iterrows():
            cidade = row.get('Cidade de Entrega', '').strip().upper()
            bairro = row.get('Bairro do Destinatario', '').strip().upper()
            if cidade == 'PORTO ALEGRE' and not df_poas.empty:
                match = df_poas[df_poas['Bairro do Destinatario'].astype(str).str.strip().str.upper() == bairro]
                if not match.empty:
                    df_processar.at[idx, 'Rota'] = match.iloc[0]['Rota']
            elif not df_rotas.empty:
                match = df_rotas[df_rotas['Cidade de Entrega'].astype(str).str.strip().str.upper() == cidade]
                if not match.empty:
                    df_processar.at[idx, 'Rota'] = match.iloc[0]['Rota']

        # --- Identificação de Entregas 'Paletizadas' ---
        try:
            paletizadores_data = supabase.table("Paletizadores").select('"CNPJ DESTINATARIO", "PALETIZADORA"').execute().data
            if paletizadores_data:
                df_paletizadores_raw = pd.DataFrame(paletizadores_data)
                df_paletizadores_raw.columns = df_paletizadores_raw.columns.str.strip().str.replace('"', '')
                if "CNPJ DESTINATARIO" not in df_paletizadores_raw.columns or "PALETIZADORA" not in df_paletizadores_raw.columns:
                    raise ValueError("Colunas 'CNPJ DESTINATARIO' ou 'PALETIZADORA' não encontradas.")
                df_paletizadores_sim = df_paletizadores_raw[
                    df_paletizadores_raw['PALETIZADORA'].astype(str).str.upper().str.strip() == 'SIM'
                ].copy()
                cnpjs_paletizadores_set = set(
                    df_paletizadores_sim['CNPJ DESTINATARIO'].astype(str).apply(lambda x: re.sub(r'\D', '', x)).unique()
                )
                df_processar['CNPJ Destinatario_limpo'] = df_processar['CNPJ Destinatario'].astype(str).apply(lambda x: re.sub(r'\D', '', x))
                df_processar['Tipo_Tratativa_Especial'] = None
                df_processar.loc[df_processar['CNPJ Destinatario_limpo'].isin(cnpjs_paletizadores_set), 'Tipo_Tratativa_Especial'] = 'Paletizadas'
                df_processar.drop(columns=['CNPJ Destinatario_limpo'], inplace=True, errors='ignore')
        except Exception as e:
            print(f"ERROR: Erro ao integrar dados de 'Paletizadores': {e}. 'Tipo_Tratativa_Especial' será nula.")
            df_processar['Tipo_Tratativa_Especial'] = None

        # Pré-roterização
        hoje = pd.to_datetime('today').normalize()
        df_processar['Codigo da Ultima Ocorrencia'] = pd.to_numeric(df_processar.get('Codigo da Ultima Ocorrencia'), errors='coerce')
        codigos_criticos = [17, 78, 79]

        obrigatorias = df_processar[
            (df_processar['Data de Embarque'].dt.normalize() < hoje + pd.Timedelta(days=1)) |
            (df_processar['Entrega Programada'].notna()) |
            (df_processar['Codigo da Ultima Ocorrencia'].isin(codigos_criticos))
        ].copy()
        confirmadas = df_processar[~df_processar['Serie_Numero_CTRC'].isin(obrigatorias['Serie_Numero_CTRC'])].copy()

        obrigatorias.drop_duplicates(subset='Serie_Numero_CTRC', inplace=True)
        confirmadas.drop_duplicates(subset='Serie_Numero_CTRC', inplace=True)

        for df_target in [obrigatorias, confirmadas]:
            if 'data_aprovacao_diretoria' in df_target.columns: df_target['data_aprovacao_diretoria'] = None
            if 'aprovador_diretoria_login' in df_target.columns: df_target['aprovador_diretoria_login'] = None
            if 'CNPJ_part' in df_target.columns: df_target.drop(columns=['CNPJ_part'], inplace=True, errors='ignore')


        colunas_finais = [
            'Serie_Numero_CTRC', 'Data de Emissao','Cliente Pagador', 'Chave CT-e', 'Cliente Destinatario',
            'Cidade de Entrega', 'Bairro do Destinatario', 'Previsao de Entrega','CNPJ Destinatario',
            'Numero da Nota Fiscal', 'Status', 'Entrega Programada', 'Particularidade',
            'Codigo da Ultima Ocorrencia', 'Peso Real em Kg', 'Peso Calculado em Kg',
            'Cubagem em m³', 'Quantidade de Volumes', 'Valor do Frete', 'Rota','Regiao',
            'CEP de Entrega','CEP do Destinatario','CEP do Remetente',
            'Tipo_Tratativa_Especial', 'is_retorno_rota'
        ]

        obrigatorias_prepared = _prepare_df_for_supabase_insert(obrigatorias[[col for col in colunas_finais if col in obrigatorias.columns]])
        confirmadas_prepared = _prepare_df_for_supabase_insert(confirmadas[[col for col in colunas_finais if col in confirmadas.columns]])

        # Inserção em lote para Dash (retorna list of dicts)
        inserted_obrigatorias = inserir_em_lote("pre_roterizacao", obrigatorias_prepared.to_dict(orient='records'))
        inserted_confirmadas = inserir_em_lote("confirmadas_producao", confirmadas_prepared.to_dict(orient='records'))

        return {
            'status': 'success',
            'message': f"Inseridos {len(obrigatorias)} NOVAS entregas em Pré Roterização e {len(confirmadas)} NOVAS em Confirmar Produção.",
            'inserted_prerot': len(obrigatorias),
            'inserted_confirm': len(confirmadas)
        }

    except Exception as e:
        return {'status': 'error', 'message': f"[ERRO] Regras de sincronização: {e}.", 'inserted_prerot': 0, 'inserted_confirm': 0}

def inserir_em_lote(nome_tabela, df_dict_list, lote=100, tentativas=3, pausa=0.2):
    """
    Insere uma lista de dicionários em lotes no Supabase.
    Retorna True em caso de sucesso total, False em caso de falha.
    """
    if not df_dict_list:
        return True # Nada para inserir

    df = pd.DataFrame(df_dict_list) # Converte de volta para DataFrame aqui para processamento interno
    dados = df.to_dict(orient="records") # Garantir que está no formato correto para Supabase

    all_success = True
    for i in range(0, len(dados), lote):
        sublote = dados[i:i + lote]
        for tentativa in range(tentativas):
            try:
                # O parâmetro on_conflict é crucial para UPSERT (INSERT OR UPDATE)
                response = supabase.table(nome_tabela).upsert(sublote, on_conflict='Serie_Numero_CTRC').execute()
                #print(f"DEBUG: Inseridos/Atualizados {len(sublote)} registros na tabela '{nome_tabela}'.")
                break # Sai do loop de tentativas se o lote for bem-sucedido
            except Exception as e:
                print(f"[TENTATIVA {tentativa + 1}] Erro ao inserir lote {i}-{i + len(sublote) - 1} na tabela '{nome_tabela}': {e}")
                time.sleep(pausa)
        else: # Se o loop de tentativas terminar sem um 'break'
            print(f"[ERRO] Falha final ao inserir lote {i}-{i + len(sublote) - 1} na tabela '{nome_tabela}'.")
            all_success = False
        time.sleep(pausa) # Pausa entre os lotes
    return all_success

def salvar_hora_sincronizacao(username):
    """Salva a data e hora da última sincronização no Supabase."""
    agora_iso = datetime.now(FUSO_BRASIL).isoformat()
    sync_info = {"timestamp": agora_iso, "username": username}
    try:
        supabase.table("metadados").upsert({"chave": "ultima_sincronizacao", "valor": json.dumps(sync_info)}, on_conflict='chave').execute()
        return True
    except Exception as e:
        print(f"Erro ao salvar hora da sincronização: {e}")
        return False

def recuperar_hora_sincronizacao():
    try:
        # Use retry_supabase_query aqui também
        dados_response = retry_supabase_query(
            lambda: supabase.table("metadados").select("valor").eq("chave", "ultima_sincronizacao").execute()
        )
        dados = dados_response # O resultado de retry_supabase_query já é o objeto de resposta

        if dados.data:
            valor_salvo = dados.data[0]["valor"]
            try:
                sync_info = json.loads(valor_salvo)
                timestamp_raw = sync_info.get("timestamp")
                username_str = sync_info.get("username", "Desconhecido")
            except json.JSONDecodeError:
                timestamp_raw = valor_salvo
                username_str = "Desconhecido (Formato Antigo)"
            
            # --- MODIFIED: Ensure timestamp_raw is always treated as a string for formatar_data_hora_br ---
            if timestamp_raw is None:
                data_hora_formatada = "N/A" # Ou algum outro placeholder
            else:
                data_hora_formatada = formatar_data_hora_br(str(timestamp_raw))
            # --- END MODIFIED ---
            
            return data_hora_formatada, username_str
        else:
            return "Não realizada", "N/A" # Retorna strings para evitar None
    except Exception as e:
        print(f"Erro ao recuperar hora da sincronização (após retries): {e}", exc_info=True)
        return "Erro", "Erro" # Retorna strings para evitar None

def salvar_historico_sincronizacao(username, num_new_fbaseroter, num_confirmadas, num_pre_roterizacao, num_paletizadas, num_novas_confirmadas, num_novas_pre_roterizacao, file_name=None, file_hash=None, detailed_update_counts=None):
    """Salva um registro de sincronização como uma nova linha na tabela 'historico_sincronizacoes'."""
    if detailed_update_counts is None:
        detailed_update_counts = {}

    agora_iso = datetime.now(FUSO_BRASIL).isoformat()

    sync_details = {
        "id": str(uuid4()), # Supabase UUID type
        "usuario": username,
        "data_hora_sincronizacao": agora_iso,
        "novas_entregas_fbaseroter": num_new_fbaseroter,
        "qtd_confirmadas_producao": num_confirmadas,
        "qtd_pre_roterizacao": num_pre_roterizacao,
        "qtd_paletizadas": num_paletizadas,
        "novas_confirmadas_producao": num_novas_confirmadas,
        "novas_pre_roterizacao": num_novas_pre_roterizacao,
        "nome_arquivo_excel": file_name,
        "arquivo_excel_hash": file_hash,
        "status_sincronizacao": "SUCESSO", # Ajuste se houver falha
        "detailed_update_counts": json.dumps(detailed_update_counts)
    }

    try:
        supabase.table("historico_sincronizacoes").insert(sync_details).execute()
        return True
    except Exception as e:
        print(f"Erro ao salvar histórico de sincronização: {e}")
        return False
