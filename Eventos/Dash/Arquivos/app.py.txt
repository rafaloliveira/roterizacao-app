@app.callback(
    Output('sync-initial-metrics-store', 'data'), # <-- Aqui o allow_duplicate NÃO é necessário, pois apenas este callback é ativado pela aba
    Input('main-tabs', 'value'), # Input: mudar para a aba de sincronização

    prevent_initial_call=True # Evita disparar na carga total da página, só quando a aba muda
)
def load_metrics_on_sync_tab_selection(tab_value, login_data):
    if tab_value == 'sincronizacao': # Só dispara se a aba 'sincronizacao' foi selecionada
        # Fetch current metrics from Supabase
        qtd_confirmadas = len(supabase.table("confirmadas_producao").select("Serie_Numero_CTRC").execute().data or [])
        qtd_pre_roterizacao = len(supabase.table("pre_roterizacao").select("Serie_Numero_CTRC").execute().data or [])
        
        qtd_paletizadas_confirmadas = len(supabase.table("confirmadas_producao").select("Serie_Numero_CTRC").eq("Tipo_Tratativa_Especial", "Paletizadas").execute().data or [])
        qtd_paletizadas_pre_roterizacao = len(supabase.table("pre_roterizacao").select("Serie_Numero_CTRC").eq("Tipo_Tratativa_Especial", "Paletizadas").execute().data or [])
        total_paletizadas = qtd_paletizadas_confirmadas + qtd_paletizadas_pre_roterizacao

        last_sync_time, last_sync_user = recuperar_hora_sincronizacao()
        
        return {
            'qtd_confirmadas': qtd_confirmadas,
            'qtd_pre_roterizacao': qtd_pre_roterizacao,
            'qtd_paletizadas': total_paletizadas,
            'last_sync_time': last_sync_time,
            'last_sync_user': last_sync_user
        }
    raise dash.exceptions.PreventUpdate # Não fazer nada se a aba não for a de sincronização

# Callback para processar o upload do arquivo
@app.callback(
    [Output('output-upload-state', 'children'),
     Output('uploaded-file-processed-data-store', 'data'),
     Output('current-uploaded-file-hash', 'data'),
     Output('start-sync-button', 'disabled', allow_duplicate=True)],
    Input('upload-data', 'contents'),
    State('upload-data', 'filename'),
    State('current-uploaded-file-hash', 'data'),
    prevent_initial_call=True # <--- ADICIONADO AQUI
)
def upload_file_and_prepare_data_callback(contents, filename, current_hash_stored):
    if contents is not None and filename:
        # Calculate hash of current uploaded content
        new_file_hash = hashlib.md5(contents.encode('utf-8')).hexdigest()

        if new_file_hash == current_hash_stored:
            # File is the same, no need to re-process, just update message
            return html.Div(f"Arquivo '{filename}' já carregado. Pronto para sincronizar.", className="text-info mt-2"), dash.no_update, dash.no_update, False

        # Process the new file
        processed_data_result = load_and_prepare_data(contents, filename)

        if processed_data_result['status'] == 'success':
            return html.Div([
                dbc.Alert(processed_data_result['message'], color="success", className="mt-2")
            ]), processed_data_result['data'], new_file_hash, False
        else:
            return html.Div([
                dbc.Alert(processed_data_result['message'], color="danger", className="mt-2")
            ]), None, None, True
    return html.Div("Aguardando o upload de um arquivo Excel válido para iniciar a sincronização.", className="text-muted mt-2"), None, None, True


# Callback principal para iniciar e gerenciar a sincronização
@app.callback(
    [Output('sync-status-messages-store', 'data'),
     Output('sync-progress-bar', 'value'),
     Output('sync-progress-bar', 'label'),
     Output('sync-progress-bar-container', 'style'),
     Output('sync-initial-metrics-store', 'data', allow_duplicate=True),
     Output('start-sync-button', 'disabled')
    ],
    Input('start-sync-button', 'n_clicks'),
    State('uploaded-file-processed-data-store', 'data'),
    State('current-uploaded-file-hash', 'data'),
    State('upload-data', 'filename'),
    State('login-status-store', 'data'),
    State('sync-initial-metrics-store', 'data'),
    prevent_initial_call=True
)
def handle_full_sync_process(n_clicks, df_to_process_data_list, file_hash, filename, login_data, initial_metrics):
    if n_clicks and df_to_process_data_list:
        username = login_data.get('username', 'Desconhecido')
        
        # Reset progress bar and show it
        progress = 0
        return_outputs = {
            'sync_status': {'message': 'Iniciando sincronização...', 'type': 'info', 'progress': 0},
            'progress_bar_value': progress,
            'progress_bar_label': f"{progress}%",
            'progress_bar_style': {'display': 'block'},
            'updated_initial_metrics': dash.no_update, # Default no_update
            'start_sync_button_disabled': True
        }
        
        # --- Stage 1: UPSERT to fBaseroter ---
        progress = 10
        return_outputs['sync_status']['message'] = "🔄 Carregando e processando arquivo em fBaseroter..."
        return_outputs['sync_status']['type'] = 'info'
        return_outputs['progress_bar_value'] = progress
        return_outputs['progress_bar_label'] = f"{progress}%"
        # Não há 'return' aqui. Apenas atualizamos os valores no dicionário.
        
        fbaseroter_upsert_status = {'message': '', 'status': 'info'}
        try:
            df_final_for_upsert = pd.DataFrame(df_to_process_data_list)

            # Filtering based on ALLOWED_OCCURRENCE_CODES (same logic as in sync_logic)
            if 'Codigo da Ultima Ocorrencia' in df_final_for_upsert.columns:
                df_final_for_upsert['Codigo da Ultima Ocorrencia'] = pd.to_numeric(
                    df_final_for_upsert['Codigo da Ultima Ocorrencia'], errors='coerce'
                )
                df_incoming_allowed = df_final_for_upsert[
                    df_final_for_upsert['Codigo da Ultima Ocorrencia'].isin(ALLOWED_OCCURRENCE_CODES)
                ].copy()
            else:
                fbaseroter_upsert_status['message'] = "WARNING: Coluna 'Codigo da Ultima Ocorrencia' não encontrada. Todas as entregas do arquivo serão consideradas permitidas para 'fBaseroter'."
                fbaseroter_upsert_status['status'] = 'warning'
                df_incoming_allowed = df_final_for_upsert.copy()

            # Ensure 'Serie_Numero_CTRC' is string and drop duplicates
            if 'Serie_Numero_CTRC' in df_incoming_allowed.columns:
                df_incoming_allowed['Serie_Numero_CTRC'] = df_incoming_allowed['Serie_Numero_CTRC'].astype(str)
                df_incoming_allowed.drop_duplicates(subset=['Serie_Numero_CTRC'], inplace=True)
            else:
                raise ValueError("Coluna 'Serie_Numero_CTRC' não encontrada no arquivo importado.")
            
            # --- Check for returning CTRCs (if they were in cargas_fechadas) ---
            existing_cargas_fechadas_data = supabase.table("cargas_fechadas").select("Serie_Numero_CTRC").execute().data
            existing_cargas_fechadas_ctrcs_set = {record["Serie_Numero_CTRC"] for record in existing_cargas_fechadas_data}
            
            returning_ctrcs_mask = df_incoming_allowed['Serie_Numero_CTRC'].isin(existing_cargas_fechadas_ctrcs_set)
            if returning_ctrcs_mask.any():
                df_incoming_allowed.loc[returning_ctrcs_mask, 'is_retorno_rota'] = True
                print(f"DEBUG: Detectado(s) {returning_ctrcs_mask.sum()} CTRC(s) retornando ao fluxo.")

            incoming_allowed_ctrcs_set = set(df_incoming_allowed['Serie_Numero_CTRC'].tolist())

            # Delete old fBaseroter records not in new upload
            existing_fbaseroter_data = supabase.table("fBaseroter").select("Serie_Numero_CTRC").execute().data
            existing_fbaseroter_ctrcs_set = {record["Serie_Numero_CTRC"] for record in existing_fbaseroter_data}
            ctrcs_to_delete = existing_fbaseroter_ctrcs_set - incoming_allowed_ctrcs_set
            if ctrcs_to_delete:
                supabase.table("fBaseroter").delete().in_("Serie_Numero_CTRC", list(ctrcs_to_delete)).execute()
                print(f"DEBUG: Deletados {len(ctrcs_to_delete)} registros de 'fBaseroter'.")
            
            # AGORA PREPARAMOS O DF AQUI ANTES DO UPSERT E ANTES DE PASSAR PARA sincronizar_fluxo_com_fbaseroter
            df_incoming_allowed_prepared = _prepare_df_for_supabase_insert(df_incoming_allowed.copy())
            records_to_upsert = df_incoming_allowed_prepared.to_dict(orient='records')

            if records_to_upsert:
                supabase.table("fBaseroter").upsert(records_to_upsert, on_conflict='Serie_Numero_CTRC').execute()
                print(f"DEBUG: UPSERT de {len(records_to_upsert)} entregas em 'fBaseroter'.")
            
            num_fbaseroter_processed = len(records_to_upsert)
            
            return_outputs['sync_status']['message'] = f"✅ {num_fbaseroter_processed} entregas processadas em 'fBaseroter'. " + fbaseroter_upsert_status['message']
            return_outputs['sync_status']['type'] = 'success' if fbaseroter_upsert_status['status'] != 'warning' else 'warning'

            return_outputs['progress_bar_value'] = progress
            return_outputs['progress_bar_label'] = f"{progress}%"
            return_outputs['start_sync_button_disabled'] = False # Re-enable button on error

        progress = 30
        return_outputs['progress_bar_value'] = progress
        return_outputs['progress_bar_label'] = f"{progress}%"
        # Não há 'return' aqui. Continuamos para o próximo estágio.

        # --- Stage 2: Synchronize Flow Tables ---
        progress = 60
        return_outputs['sync_status']['message'] = "🔄 Sincronizando tabelas do fluxo com fBaseroter..."
        return_outputs['sync_status']['type'] = 'info'
        return_outputs['progress_bar_value'] = progress
        return_outputs['progress_bar_label'] = f"{progress}%"
        # Não há 'return' aqui.

        # PASSA O DATAFRAME JÁ PREPARADO `df_incoming_allowed_prepared`
        sync_flow_result = sincronizar_fluxo_com_fbaseroter(df_incoming_allowed_prepared)

def reset_uploader_and_button(n_clicks_refresh, sync_status, login_data):

        raise dash.exceptions.PreventUpdate

        qtd_pre_roterizacao = 0

                response_prerot = retry_supabase_query(
                    lambda: supabase.table("pre_roterizacao").select("Serie_Numero_CTRC").execute()

# Callback para baixar o histórico de sincronizações
@app.callback(
    Output('download-sync-history-data', 'data'),
    Input('download-sync-history-button', 'n_clicks'),
    prevent_initial_call=True
)
def generate_sync_history_csv(n_clicks):
    if n_clicks:
        try:
            response = supabase.table("historico_sincronizacoes").select("*").order("data_hora_sincronizacao", desc=True).execute()
            historico_data = response.data
            
            if not historico_data:
                return dash.no_update # Or return a message that no data is available
            
            df_historico = pd.DataFrame(historico_data)
            
            if 'data_hora_sincronizacao' in df_historico.columns:
                df_historico['data_hora_sincronizacao'] = pd.to_datetime(df_historico['data_hora_sincronizacao'], utc=True)
                df_historico['data_hora_sincronizacao'] = df_historico['data_hora_sincronizacao'].apply(formatar_data_hora_br)

            if 'detailed_update_counts' in df_historico.columns:
                df_historico['detailed_update_counts_parsed'] = df_historico['detailed_update_counts'].apply(
                    lambda x: json.loads(x) if pd.notna(x) and x else {}
                )
                df_historico['Detalhes das Atualizações'] = df_historico['detailed_update_counts_parsed'].apply(
                    lambda d: ", ".join([f"{k.replace('_', ' ').title()}: {v}" for k, v in d.items()]) if d else "Nenhum"
                )
                df_historico.drop(columns=['detailed_update_counts', 'detailed_update_counts_parsed'], inplace=True, errors='ignore')

            df_historico.rename(columns={
                'data_hora_sincronizacao': 'Data/Hora Sinc.',
                'usuario': 'Usuario',
                'novas_entregas_fbaseroter': 'Novas Entregas fBaseroter',
                'qtd_confirmadas_producao': 'Entregas em Confirmar Producao (Total)',
                'qtd_pre_roterizacao': 'Entregas em Pre-Roterizacao (Total)',
                'qtd_paletizadas': 'Entregas Paletizadas (Total)',
                'novas_confirmadas_producao': 'Novas em Confirmar Producao (nesta Sync)',
                'novas_pre_roterizacao': 'Novas em Pre-Roterizacao (nesta Sync)',
                'nome_arquivo_excel': 'Nome do Arquivo Excel',
                'arquivo_excel_hash': 'Hash do Arquivo',
                'status_sincronizacao': 'Status'
            }, inplace=True)

            colunas_csv_ordenadas = [
                'Data/Hora Sinc.', 'Usuario', 'Status',
                'Novas Entregas fBaseroter',
                'Novas em Confirmar Producao (nesta Sync)',
                'Novas em Pré-Roterizacao (nesta Sync)',
                'Detalhes das Atualizações',
                'Entregas em Confirmar Producao (Total)',
                'Entregas em Pre-Roterizacao (Total)',
                'Entregas Paletizadas (Total)',
                'Nome do Arquivo Excel', 'Hash do Arquivo'
            ]
            colunas_csv_ordenadas_existentes = [col for col in colunas_csv_ordenadas if col in df_historico.columns]
            df_historico = df_historico[colunas_csv_ordenadas_existentes]

            csv_string = df_historico.to_csv(index=False, sep=';', encoding='utf-8-sig')
            
            return dcc.send_string(csv_string, "historico_sincronizacoes.csv")
        
        except Exception as e:
            print(f"Erro ao gerar histórico de sincronização CSV: {e}")
            return dash.no_update # Or an alert to the user

@app.callback(

    Input('sync-status-messages-store', 'data')
)
def display_sync_flow_updates(sync_status):
    # Only try to fetch details if a sync has just completed successfully
    if sync_status and sync_status['type'] == 'success' and sync_status['progress'] == 100:
        try:
            last_history_record = supabase.table("historico_sincronizacoes") \
                .select("detailed_update_counts") \
                .order("data_hora_sincronizacao", desc=True) \
                .limit(1) \
                .execute()
            
            if last_history_record.data and last_history_record.data[0].get("detailed_update_counts"):
                detailed_updates = json.loads(last_history_record.data[0]["detailed_update_counts"])
                
                if detailed_updates:
                    update_list_items = []
                    for table_name, count in detailed_updates.items():
                        display_name = TABLE_TO_PAGE_NAME_MAP.get(table_name, table_name.replace('_', ' ').title())
                        update_list_items.append(html.Li(f"- {display_name}: {count} entregas"))
                    
                    return html.Div([
                        html.H5("Entregas Atualizadas/Afetadas nas Etapas do Fluxo (Última Sincronização):"),
                        html.Ul(update_list_items)
                    ])
                else:
                    return dbc.Alert("Nenhum detalhe de atualização específico encontrado para as tabelas do fluxo na última sincronização (ou todas as entregas já estavam atualizadas).", color="info")
            else:
                return dbc.Alert("Nenhum detalhe de atualização de sincronização anterior para exibir.", color="info")
        except Exception as e:
            return dbc.Alert(f"Erro ao carregar detalhes de atualização da sincronização: {e}", color="danger")

# Callback para Alterar Senha
@app.callback(
    [Output('change-password-output', 'children'),
     Output('current-page-route-store', 'data', allow_duplicate=True),
     Output('login-status-store', 'data', allow_duplicate=True)],
    Input('update-password-button', 'n_clicks'),
    State('current-password-input', 'value'),
    State('new-password-input', 'value'),
    State('confirm-new-password-input', 'value'),
    State('login-status-store', 'data'),
    prevent_initial_call=True
)
def update_password(n_clicks, current_password, new_password, confirm_new_password, login_data):
    if n_clicks:
        username = login_data.get("username")
        if not username:
            return dbc.Alert("Usuário não autenticado.", color="danger"), dash.no_update, dash.no_update

        if not current_password or not new_password or not confirm_new_password:
            return dbc.Alert("⚠️ Por favor, preencha todos os campos.", color="warning"), dash.no_update, dash.no_update

        if new_password != confirm_new_password:
            return dbc.Alert("⚠️ A nova senha e a confirmação não coincidem.", color="warning"), dash.no_update, dash.no_update

        user_auth = autenticar_usuario_dash(username, current_password)
        if not user_auth:
            return dbc.Alert("❌ Senha atual incorreta.", color="danger"), dash.no_update, dash.no_update

        try:
            new_hash = hash_senha(new_password)
            update_data = {"senha_hash": new_hash, "precisa_alterar_senha": False} # Remove a flag

            supabase.table("usuarios").update(update_data).eq("nome_usuario", username).execute()
            
            # Força o logout para o usuário fazer login com a nova senha
            return dbc.Alert("✅ Senha alterada com sucesso! Faça login novamente com a nova senha.", color="success"), 'login', {'logged_in': False, 'username': None, 'is_admin': False, 'classe': 'colaborador'}
        except Exception as e:
            return dbc.Alert(f"❌ Erro ao atualizar senha: {e}", color="danger"), dash.no_update, dash.no_update
    return dash.no_update, dash.no_update, dash.no_update
