date_only_formatter_dash = {
    'function': '''
        function(params) {
            if (params.value === null || typeof params.value === 'undefined' || params.value === '') {
                return '';
            }
            
            let dateStr = String(params.value);
            let datePart = dateStr.split('T')[0];
            let parts = datePart.split('-');
            
            if (parts.length === 3) {
                let year = parts[0];
                let month = parts[1].padStart(2, '0');
                let day = parts[2].padStart(2, '0');
                return day + '-' + month + '-' + year;
            }
            
            return datePart;
        }
    '''
}

@callback(
    Output('confirmadas-producao-data-store', 'data'),
    Output('aprovacao-diretoria-data-store', 'data'),
    # Input que serve para forçar o recarregamento dos dados
    Input('confirmadas-producao-reload-flag', 'data'),
    prevent_initial_call=False # Permite que este callback seja disparado na carga inicial da página
)
def load_confirmadas_producao_data(reload_flag):
    df_confirmadas_data = pd.DataFrame()
    df_aprovacao_data = pd.DataFrame()

    try:
        # Carregar df_confirmadas com retry
        logger.info("Tentando buscar dados de 'confirmadas_producao' com retry.")
        confirmadas_response = retry_supabase_query(
            lambda: supabase.table("confirmadas_producao").select("*").execute()
        )
        confirmadas_raw = confirmadas_response.data # Obter os dados da resposta
        df_confirmadas_data = pd.DataFrame(confirmadas_raw)

        # Adicione esta correção:
        if not df_confirmadas_data.empty:
            # Corrige datas que podem ter formato incorreto (ex: 2025-11-010 -> 2025-11-10)
            date_columns_to_fix = ['Data de Emissao', 'Previsao de Entrega', 'Entrega Programada']
            for col in date_columns_to_fix:
                if col in df_confirmadas_data.columns:
                # Força conversão para datetime e depois para string ISO limpa
                    df_confirmadas_data[col] = pd.to_datetime(df_confirmadas_data[col], errors='coerce')
                    df_confirmadas_data[col] = df_confirmadas_data[col].dt.strftime('%Y-%m-%d')

        if not df_confirmadas_data.empty:
            # --- Correção de Tipos e Formatação para JSON ---

            # Aplicar _parse_date_robustly para TODAS as colunas de data/timestamp relevantes
            # ANTES de tentar isoformat().
            # 'Entrega Programada' pode vir como string ou datetime, então também é tratada aqui.
            all_date_cols_to_process = list(set(GLOBAL_DB_TIMESTAMP_COLS + GLOBAL_DB_DATE_COLS + ['Entrega Programada']))
            for col in all_date_cols_to_process:
                if col in df_confirmadas_data.columns:
                    # Primeiro, converte a coluna para objetos datetime
                    df_confirmadas_data[col] = _parse_date_robustly(df_confirmadas_data[col], col_name=col)
                    # Agora que são objetos datetime ou NaT, podemos chamar .isoformat()
                    df_confirmadas_data[col] = df_confirmadas_data[col].apply(
                        lambda x: x.isoformat() if pd.notna(x) else None
                    )

            # As outras colunas que são strings ou numéricas devem ser tratadas como já eram
            if 'Rota' in df_confirmadas_data.columns:
                df_confirmadas_data['Rota'] = df_confirmadas_data['Rota'].fillna('').astype(str)
            if 'Status' in df_confirmadas_data.columns:
                df_confirmadas_data['Status'] = df_confirmadas_data['Status'].fillna('').astype(str)
            # 'Entrega Programada' já foi tratada acima no loop genérico.
            if 'Particularidade' in df_confirmadas_data.columns:
                df_confirmadas_data['Particularidade'] = df_confirmadas_data['Particularidade'].fillna('').astype(str)
            if 'Serie_Numero_CTRC' in df_confirmadas_data.columns:
                df_confirmadas_data['Serie_Numero_CTRC'] = df_confirmadas_data['Serie_Numero_CTRC'].astype(str)
            if 'Cliente Pagador' in df_confirmadas_data.columns:
                df_confirmadas_data['Cliente Pagador'] = df_confirmadas_data['Cliente Pagador'].fillna('').astype(str)
            if 'Tipo_Tratativa_Especial' not in df_confirmadas_data.columns:
                df_confirmadas_data['Tipo_Tratativa_Especial'] = ''
            df_confirmadas_data['Tipo_Tratativa_Especial'] = df_confirmadas_data['Tipo_Tratativa_Especial'].fillna('').astype(str)


        # Carregar df_aprovacao (para as métricas) com retry
        logger.info("Tentando buscar dados de 'aprovacao_diretoria' com retry.")
        aprovacao_response = retry_supabase_query(
            lambda: supabase.table("aprovacao_diretoria").select("*").execute()
        )
        aprovacao_raw = aprovacao_response.data
        df_aprovacao_data = pd.DataFrame(aprovacao_raw)
        if not df_aprovacao_data.empty:
            # Garante que as colunas numéricas existam e sejam numéricas para soma
            numeric_cols = ["Peso Real em Kg", "Peso Calculado em Kg", "Valor do Frete", "Cubagem em m³", "Quantidade de Volumes"]
            for col in numeric_cols:
                if col in df_aprovacao_data.columns:
                    df_aprovacao_data[col] = pd.to_numeric(df_aprovacao_data[col], errors='coerce').fillna(0)
            
            # Formata colunas de data/timestamp para JSON string (se houver, seguindo a mesma lógica)
            all_date_cols_aprovacao = list(set(GLOBAL_DB_TIMESTAMP_COLS + GLOBAL_DB_DATE_COLS))
            for col in all_date_cols_aprovacao:
                if col in df_aprovacao_data.columns:
                    df_aprovacao_data[col] = _parse_date_robustly(df_aprovacao_data[col], col_name=col)
                    df_aprovacao_data[col] = df_aprovacao_data[col].apply(
                        lambda x: x.isoformat() if pd.notna(x) else None
                    )

        logger.info("Dados de 'confirmadas_producao' e 'aprovacao_diretoria' carregados com sucesso.")
    except Exception as e:
        logger.error(f"Erro ao carregar dados do Supabase na página de Confirmar Produção: {e}", exc_info=True)
        return json.dumps(pd.DataFrame().to_dict('records')), json.dumps(pd.DataFrame().to_dict('records'))

    # Retorna os dados em formato JSON (lista de dicionários)
    return json.dumps(df_confirmadas_data.to_dict('records')), json.dumps(df_aprovacao_data.to_dict('records'))
