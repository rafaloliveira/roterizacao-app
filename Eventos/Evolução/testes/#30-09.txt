                                st.session_state['last_action_message_type'] = 'success'
                                st.session_state['last_action_message_text'] = f"✅ Informações da carga {carga} salvas com sucesso!"
                                st.rerun() # Força recarregamento da página para refletir as mudanças.

                        except Exception as e:
                            st.session_state['last_action_message_type'] = 'error'
                            st.session_state['last_action_message_text'] = f"❌ Erro ao salvar dados da carga {carga}: {e}"
                            st.rerun() # Exibe o erro e força recarregamento.

                    # --- 5. BOTÃO "ENVIAR PARA APROVAÇÃO DE CUSTOS" ---
                    btn_aprovar_custos_key = f"btn_aprov_custos_{carga}"
                    if st.button(
                        f"➤ Enviar para Aprovação de Custos",
                        key=btn_aprovar_custos_key,
                        disabled=selecionadas.empty or (valor_contratacao <= 0) # Desabilitado se sem seleção ou valor <= 0
                    ):
                        if valor_contratacao <= 0:
                            st.warning("Por favor, insira um valor de contratação válido (maior que zero).")
                        else:
                            try:
                                with st.spinner("Enviando entregas para aprovação de custos..."):
                                    # Extrai CTRCs das entregas selecionadas na AgGrid.
                                    ctrcs_selecionados_str = [s.get("Serie_Numero_CTRC") for s in selecionadas.to_dict(orient="records") if s.get("Serie_Numero_CTRC")]
                                    df_aprovar_custos = df_carga_raw[df_carga_raw['Serie_Numero_CTRC'].isin(ctrcs_selecionados_str)].copy()

                                    # Limpa colunas auxiliares/específicas do AgGrid
                                    df_aprovar_custos = df_aprovar_custos.drop(columns=["_selectedRowNodeInfo"], errors="ignore")

                                    # Zera o motivo de rejeição, se existir, para aprovação.
                                    if "motivo_rejeicao" in df_aprovar_custos.columns:
                                        df_aprovar_custos["motivo_rejeicao"] = None

                                    df_aprovar_custos["numero_carga"] = carga # Garante que a carga está associada.

                                    # Atribui os valores selecionados/inseridos na UI ao DataFrame a ser enviado.
                                    df_aprovar_custos["motorista"] = motorista_to_save
                                    df_aprovar_custos["placa"] = placa_to_save
                                    df_aprovar_custos["veiculo"] = veiculo_to_save
                                    df_aprovar_custos["valor_contratacao"] = valor_contratacao
                                    df_aprovar_custos["valor_adicional_frete"] = valor_adicional_frete
                                    df_aprovar_custos["motivo_valor_adicional"] = motivo_valor_adicional if valor_adicional_frete > 0 else None

                                    # Prepara o DataFrame para inserção no Supabase (tratamento de datas).
                                    df_aprovar_custos = _prepare_df_for_supabase_insert(df_aprovar_custos)
                                    registros_para_custos = df_aprovar_custos.to_dict(orient="records")

                                    if registros_para_custos:
                                        # Insere as entregas selecionadas na tabela de aprovação de custos.
                                        supabase.table("aprovacao_custos").insert(registros_para_custos).execute()
                                        chaves_para_remover = [r.get("Serie_Numero_CTRC") for r in registros_para_custos if r.get("Serie_Numero_CTRC")]
                                        if chaves_para_remover:
                                            # Remove as entregas aprovadas da tabela de cargas geradas.
                                            supabase.table("cargas_geradas").delete().in_("Serie_Numero_CTRC", chaves_para_remover).execute()
                                            # Atualiza os dados da carga (motorista, valores, etc.) nas entregas remanescentes.
                                            supabase.table("cargas_geradas").update({
                                                "motorista": motorista_to_save,
                                                "placa": placa_to_save,
                                                "veiculo": veiculo_to_save,
                                                "valor_contratacao": valor_contratacao,
                                                "valor_adicional_frete": valor_adicional_frete,
                                                "motivo_valor_adicional": motivo_valor_adicional if valor_adicional_frete > 0 else None
                                            }).eq("numero_carga", carga).execute()

                                        # Invalida caches e sinaliza recarregamento de ambas as páginas envolvidas.
                                        st.session_state["reload_cargas_geradas"] = True
                                        st.session_state["reload_aprovacao_custos"] = True
                                        st.session_state.pop(grid_key_id, None) # Limpa o estado do AgGrid.

                                        st.session_state['last_action_message_type'] = 'success'
                                        st.session_state['last_action_message_text'] = f"✅ {len(registros_para_custos)} entregas da carga {carga} enviadas para Aprovação de Custos com valor R\$ {valor_contratacao:.2f}."
                                        st.rerun() # Força recarregamento da página.
                                    else:
                                        st.warning("Nenhuma entrega válida selecionada para enviar para aprovação de custos.")
                            except Exception as e:
                                st.session_state['last_action_message_type'] = 'error'
                                st.session_state['last_action_message_text'] = f"❌ Erro ao enviar entregas para aprovação de custos: {e}"
                                st.rerun() # Exibe o erro e força recarregamento.
                                
    except Exception as e: # <-- Adicione um bloco except como este
        print(f"Ocorreu um erro: {e}")
